<canvas id="pieChart" style="height:400px"></canvas>
<script>
    var chartDom = document.getElementById('pieChart');
    var myChart = require('echarts').init(chartDom);
    var option;

    option = {
        title: {
            text: '物品数量分布',
            left: 'center'
        },
        tooltip: {
            trigger: 'item',
            formatter: '{a} <br/>{b}: {c} ({d}%)'
        },
        legend: {
            orient: 'vertical',
            left: 'left',
            data: ['a', 'b', 'c', 'd']
        },
        series: [
            {
                name: '物品',
                type: 'pie',
                radius: '50%',
                data: [
                    {value: 100, name: 'a'},
                    {value: 105, name: 'b'},
                    {value: 110, name: 'c'},
                    {value: 98, name: 'd'}
                ],
                emphasis: {
                    itemStyle: {
                        shadowBlur: 10,
                        shadowOffsetX: 0,
                        shadowColor: 'rgba(0, 0, 0, 0.5)'
                    }
                }
            }
        ]
    };
    option && myChart.setOption(option);
</script>
